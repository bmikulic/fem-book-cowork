%% Mathematics
\usepackage{amsmath}
\usepackage{amssymb}
\usepackage{amsthm}
\usepackage{mathtools}

    keywordstyle=\color{matlabblue}\bfseries,

\newcommand{\mass}{\matr{M}}             % global mass
\newcommand{\damp}{\matr{C}}             % global damping
\newcommand{\transpose}{^{\mathsf{T}}}   % transpose
\newcommand{\inv}{^{-1}}                 % inverse
\newcommand{\dd}{\,\mathrm{d}}           % differential
